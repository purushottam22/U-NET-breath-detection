def unet_1d(input_shape, num_classes):
    inputs = Input(shape=input_shape)

    # Encoder
    conv1 = Conv1D(64, 3, activation='relu', padding='same')(inputs)
    conv1 = Conv1D(64, 3, activation='relu', padding='same')(conv1)
    pool1 = MaxPooling1D(2)(conv1)

    conv2 = Conv1D(128, 3, activation='relu', padding='same')(pool1)
    conv2 = Conv1D(128, 3, activation='relu', padding='same')(conv2)
    pool2 = MaxPooling1D(2)(conv2)

    # Bottleneck
    conv3 = Conv1D(256, 3, activation='relu', padding='same')(pool2)
    conv3 = Conv1D(256, 3, activation='relu', padding='same')(conv3)

    # Decoder
    up2 = UpSampling1D(2)(conv3)
    up2 = Concatenate()([up2, conv2])
    conv4 = Conv1D(128, 3, activation='relu', padding='same')(up2)
    conv4 = Conv1D(128, 3, activation='relu', padding='same')(conv4)

    up1 = UpSampling1D(2)(conv4)
    up1 = Concatenate()([up1, conv1])
    conv5 = Conv1D(64, 3, activation='relu', padding='same')(up1)
    conv5 = Conv1D(64, 3, activation='relu', padding='same')(conv5)

    # Output
    outputs = Conv1D(num_classes, 1, activation='softmax')(conv5)

    return Model(inputs, outputs)
